Lead & KYC Integration Guide

Official TSS Studio blueprint for connecting Winity Life forms to a live Google Sheet and Didit secure biometric KYC verification in real time.

1. System Architecture

The integration utilizes a lightweight, zero-maintenance, and secure backend router using Google Apps Script tied directly to a Google Sheet. Leads are captured immediately when Step 1 is finished, ensuring you retain high-intent contacts even if they abandon the KYC verification step.

1. Frontend Form

Visitor fills Name, Email, Jurisdiction and submits

2. Apps Script Webhook

HTTP POST sends payload securely to Google Apps Script

3. Live Google Sheet

Row appended immediately with "KYC Pending" status

4. Didit KYC API

Secure server session created and liveness check initiated

2. Setting Up the Live Google Sheet Webhook

To connect your forms to a live spreadsheet, follow these steps to deploy the Apps Script Webhook:

  1. Create a blank Google Sheet and name it Winity Leads.
  2. Click Extensions > Apps Script in the top menu.
  3. Delete any default placeholder code and paste the production-grade script below.
  4. Click Deploy (top right) > New deployment. Select type Web app.
  5. Configure: Execute as Me, Who has access: Anyone. Deploy and copy the Web app URL.
/**
 * WINITY LIFE — REAL-TIME GOOGLE SHEETS & DIDIT KYC WEBHOOK
 * Tim Stewart Shaw | May 2026
 */
function doPost(e) {
  var corsHeaders = {
    "Access-Control-Allow-Origin": "*",
    "Access-Control-Allow-Methods": "POST, GET, OPTIONS",
    "Access-Control-Allow-Headers": "Content-Type"
  };
  
  if (e.parameter.options === 'OPTIONS') {
    return ContentService.createTextOutput("").setMimeType(ContentService.MimeType.TEXT).setHeaders(corsHeaders);
  }

  try {
    var json = JSON.parse(e.postData.contents);
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Timestamp", "First Name", "Last Name", "Email", "Jurisdiction", "Tier", "KYC Status", "Didit Session ID", "Didit Session URL"]);
    }
    
    var email = json.email;
    var firstName = json.firstName;
    var lastName = json.lastName;
    var region = json.region;
    var tier = json.tier;
    var kycStatus = json.kycStatus || "Form Initiated";
    var diditSessionId = json.diditSessionId || "";
    var diditSessionUrl = json.diditSessionUrl || "";
    
    var data = sheet.getDataRange().getValues();
    var rowIndex = -1;
    for (var i = 1; i < data.length; i++) {
      if (data[i][3] === email) {
        rowIndex = i + 1;
        break;
      }
    }
    
    var timestamp = new Date();
    var liveKyc = json.liveKyc === true || json.liveKyc === "true";
    if (liveKyc && !diditSessionUrl) {
      var diditResult = createDiditSession(firstName, lastName, email);
      if (diditResult && diditResult.session_url) {
        diditSessionUrl = diditResult.session_url;
        diditSessionId = diditResult.id;
        kycStatus = "KYC Pending (Didit)";
      }
    }
    
    if (rowIndex > -1) {
      sheet.getRange(rowIndex, 1).setValue(timestamp);
      sheet.getRange(rowIndex, 2).setValue(firstName);
      sheet.getRange(rowIndex, 3).setValue(lastName);
      sheet.getRange(rowIndex, 5).setValue(region);
      sheet.getRange(rowIndex, 6).setValue(tier);
      sheet.getRange(rowIndex, 7).setValue(kycStatus);
      if (diditSessionId) sheet.getRange(rowIndex, 8).setValue(diditSessionId);
      if (diditSessionUrl) sheet.getRange(rowIndex, 9).setValue(diditSessionUrl);
    } else {
      sheet.appendRow([timestamp, firstName, lastName, email, region, tier, kycStatus, diditSessionId, diditSessionUrl]);
    }
    
    var response = {
      success: true,
      message: "Lead synchronized successfully",
      sessionUrl: diditSessionUrl,
      sessionId: diditSessionId,
      kycStatus: kycStatus
    };
    
    return ContentService.createTextOutput(JSON.stringify(response))
      .setMimeType(ContentService.MimeType.JSON)
      .setHeaders(corsHeaders);
      
  } catch (error) {
    var errResponse = { success: false, error: error.toString() };
    return ContentService.createTextOutput(JSON.stringify(errResponse))
      .setMimeType(ContentService.MimeType.JSON)
      .setHeaders(corsHeaders);
  }
}

function doOptions(e) {
  var corsHeaders = {
    "Access-Control-Allow-Origin": "*",
    "Access-Control-Allow-Methods": "POST, GET, OPTIONS",
    "Access-Control-Allow-Headers": "Content-Type"
  };
  return ContentService.createTextOutput("").setMimeType(ContentService.MimeType.TEXT).setHeaders(corsHeaders);
}

function createDiditSession(firstName, lastName, email) {
  var DIDIT_API_KEY = "YOUR_DIDIT_API_KEY_HERE"; 
  var url = "https://verification.didit.me/v3/sessions";
  
  var payload = {
    "vendor_session_id": "winity_" + new Date().getTime(),
    "features": ["document_verification", "liveness_detection"],
    "user_data": {
      "first_name": firstName,
      "last_name": lastName,
      "email": email
    },
    "redirect_url": "https://winity.life/kyc-callback" 
  };
  
  var options = {
    "method": "post",
    "contentType": "application/json",
    "headers": {
      "x-api-key": DIDIT_API_KEY
    },
    "payload": JSON.stringify(payload),
    "muteHttpExceptions": true
  };
  
  try {
    var response = UrlFetchApp.fetch(url, options);
    var resJson = JSON.parse(response.getContentText());
    return resJson;
  } catch (err) {
    return null;
  }
}

3. Configuring the Deliverables

Paste your deployed Web App URL into the WINITY_CONFIG block inside Winity Life static HTML files (located near the top of the main script tag): 

var WINITY_CONFIG = {
  // Google Sheets Webhook URL (Apps Script Web App)
  googleSheetWebhookUrl: "https://script.google.com/macros/s/YOUR_URL_HERE/exec",

  // Didit KYC Integration Mode ("simulation" or "live")
  kycMode: "simulation" // Toggle to "live" when you plug in your API key
};

4. Configuring Live Didit KYC Key

1. Sign up or log in at business.didit.me.
2. Go to API & Webhooks and generate a business API Key.
3. Replace "YOUR_DIDIT_API_KEY_HERE" in your Google Apps Script with your API Key.
4. In your HTML deliverable file configuration block, change kycMode from "simulation" to "live". Save and you are ready!
